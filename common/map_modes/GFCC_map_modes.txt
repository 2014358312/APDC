scripted_map_modes = {
	GFCC_area_Pollution_map_mode = {
		top = { type = none }
		bottom = {
			type = state
			color = {
				set_temp_variable = { GFCC_area_Pollution_color_var = FROM.GFCC_area_Pollution_var }
				if = { limit = { check_variable = { GFCC_area_Pollution_color_var = 0 } } #白区
					set_temp_variable = { r_clc = 255 }
					set_temp_variable = { g_clc = 255 }
					set_temp_variable = { b_clc = 255 }
				}
				else_if = { 
					limit = { 
						check_variable = { GFCC_area_Pollution_color_var > 0 }
						check_variable = { GFCC_area_Pollution_color_var < 11 } 
					} #绿区
					set_temp_variable = { r_clc = 0 }
					set_temp_variable = { g_clc = 139 }
					set_temp_variable = { b_clc = 0 }
				}
				else_if = { 
					limit = { 
						check_variable = { GFCC_area_Pollution_color_var > 10 }
						check_variable = { GFCC_area_Pollution_color_var < 150 }
						FROM = { GFCC_area_Pollution_B1 < 1 }
					} #黄区
					set_temp_variable = { r_clc = 238 }
					set_temp_variable = { g_clc = 238 }
					set_temp_variable = { b_clc = 0 }
				}
				else_if = { 
					limit = {
						check_variable = { GFCC_area_Pollution_color_var > 10 }
						check_variable = { GFCC_area_Pollution_color_var < 150 }
						FROM = { GFCC_area_Pollution_B1 > 0 }
						# has_dynamic_modifier = { modifier = SLE_OGAS_diqu_2 }
					} #蓝区
					set_temp_variable = { r_clc = 100 }
					set_temp_variable = { g_clc = 149 }
					set_temp_variable = { b_clc = 237 }
				}
				else_if = { 
					limit = { 
						check_variable = { GFCC_area_Pollution_color_var > 149 }
						check_variable = { GFCC_area_Pollution_color_var < 10000 }
					} #红区
					set_temp_variable = { r_clc = 139 }
					set_temp_variable = { g_clc = 26 }
					set_temp_variable = { b_clc = 26 }
				}
				else_if = { limit = { check_variable = { GFCC_area_Pollution_color_var > 9999 } } #黑区
					set_temp_variable = { r_clc = 0 }
					set_temp_variable = { g_clc = 0 }
					set_temp_variable = { b_clc = 0 }
				}

				set_temp_variable = { red = r_clc }
				set_temp_variable = { green = g_clc }
				set_temp_variable = { blue = b_clc }
				divide_temp_variable = { red = 255 }
				divide_temp_variable = { blue = 255 }
				divide_temp_variable = { green = 255 }
				set_temp_variable = { alpha = 1.00 }
			}
		}

		far_text = state
		near_text = state

		update_monthly = yes
	}
