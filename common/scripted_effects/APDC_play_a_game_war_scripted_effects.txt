
APDC_game_war_east_start = {
	set_global_flag = apdc_game_war_starts
	every_country = {
		limit = {
			OR = {
				has_country_flag = APDC_game_participant_east
				has_country_flag = APDC_game_participant_west
			}
		}
		hidden_effect = {
			add_dynamic_modifier = {
				modifier = APDC_war_weariness
			}
		}
	}
	GER = {
		create_faction = APDC_game_east_faction
		add_to_faction = POL
		add_to_faction = ROM
		add_to_faction = YUG
		add_to_faction = CZE
		add_to_faction = BUL
		add_to_faction = HUN
	}
	FRA = {
		create_faction = APDC_game_west_faction
		add_to_faction = ENG
		add_to_faction = SPR
		add_to_faction = LUX
	}
	GER = {
		declare_war_on = {
			target = FRA
			type = puppet_wargoal_focus
		}
	}
}

APDC_game_brawl_starts = {
	USA = {
		create_faction = APDC_game_west_faction
	}
	every_country = {
		limit = {
			has_country_flag = APDC_game_participant_east
		}
		SOV = {
			add_to_faction = PREV
		}
	}
	every_country = {
		limit = {
			has_country_flag = APDC_game_participant_west
		}
		USA = {
			add_to_faction = PREV
		}
	}
	SOV = {
		declare_war_on = {
			target = USA
			type = puppet_wargoal_focus
		}
	}
	every_country = {
		limit = {
			OR = {
				has_country_flag = APDC_game_participant_east
				has_country_flag = APDC_game_participant_west
				tag = USA
				tag = SOV
			}
		}
		hidden_effect = {
			add_dynamic_modifier = {
				modifier = APDC_war_weariness
			}
		}
	}
	set_global_flag = apdc_game_war_starts
	news_event = APDC_game_war.6
}
