
APDC_game_start_button_effect = {
	if = {
		limit = { has_country_flag = APDC_game_start }
			clr_country_flag = APDC_game_start
			set_variable = { APDC_game_action = 0 }
		}
	else = {
		set_country_flag =  APDC_game_start
	}
}
APDC_game_button_icon_update = {
	set_temp_variable = {
		var = temp1
		value = APDC_USA_tendency
	}
	subtract_from_temp_variable = {
		var = temp1
		value = APDC_SOV_tendency
	}
	if = {
		limit = {
			check_variable = {
				var = temp1
				value = -14
				compare = less_than
			}
		}
		set_variable = { APDC_game_button_frame = 1 }
	}
	else_if = {
		limit = {
			check_variable = {
				var = temp1
				value = -10
				compare = less_than
			}
		}
		set_variable = { APDC_game_button_frame = 2 }
	}
	else_if = {
		limit = {
			check_variable = {
				var = temp1
				value = -6
				compare = less_than
			}
		}
		set_variable = { APDC_game_button_frame = 3 }
	}
	else_if = {
		limit = {
			check_variable = {
				var = temp1
				value = -2
				compare = less_than
			}
		}
		set_variable = { APDC_game_button_frame = 4 }
	}
	
	else_if = {
		limit = {
			check_variable = {
				var = temp1
				value = 2
				compare = greater_than
			}
		}
		set_variable = { APDC_game_button_frame = 6 }
	}
	else_if = {
		limit = {
			check_variable = {
				var = temp1
				value = 6
				compare = greater_than
			}
		}
		set_variable = { APDC_game_button_frame = 7 }
	}
	else_if = {
		limit = {
			check_variable = {
				var = temp1
				value = 10
				compare = greater_than
			}
		}
		set_variable = { APDC_game_button_frame = 8 }
	}
	else_if = {
		limit = {
			check_variable = {
				var = temp1
				value = 14
				compare = greater_than
			}
		}
		set_variable = { APDC_game_button_frame = 9 }
	}
	else = {
		set_variable = { APDC_game_button_frame = 5 }
	}
}

APDC_game_faction_judge = {
	every_country = {
		limit = { has_variable = APDC_game_button_frame }
		if = {
			limit = {
				check_variable = {
					APDC_game_button_frame < 5
				}
			}
			clr_country_flag = APDC_game_participant_west
			set_country_flag = APDC_game_participant_east
		}
		else_if = {
			limit = {
				check_variable = {
					APDC_game_button_frame = 5
				}
			}
			clr_country_flag = APDC_game_participant_west
		}
	}

}
