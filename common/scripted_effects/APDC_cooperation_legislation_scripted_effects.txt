APDC_initialize_cooperation_legislation = {
	if = {

		# v2 deliberately reinitializes saves touched by the first legislation
		# build, whose unsupported loop syntax left all arrays empty.

		clear_array = global.APDC_cooperation_law_selected_options

		# its stable numeric id, so adding later groups does not rewrite GUI.

			limit = { check_variable = { APDC_cooperation_init_index < 41 } }
			add_to_array = { array = global.APDC_cooperation_law_option_counts value = 4 }

# Migrates an in-progress ballot or recess from the former generic daily hook.
# It is safe to call both at startup and when the panel is opened.
APDC_repair_cooperation_timer_daily_rate = {
	if = {
		limit = {
			NOT = { has_global_flag = APDC_cooperation_timer_daily_rate_fixed }
		}
		if = {
			limit = {
				OR = {
					check_variable = { global.APDC_cooperation_vote_state = 1 }
					check_variable = { global.APDC_cooperation_vote_state = 2 }
				}
			}
			set_variable = { global.APDC_cooperation_vote_days_remaining = 30 }
		}
		set_global_flag = APDC_cooperation_timer_daily_rate_fixed
	}
}

APDC_start_next_cooperation_law_vote = {
	if = {
		limit = {
			check_variable = { global.APDC_cooperation_vote_state = 0 }
			check_variable = { global.APDC_cooperation_law_queue_groups^num > 0 }
		}
		set_variable = { global.APDC_cooperation_active_law_group = global.APDC_cooperation_law_queue_groups^0 }
		set_variable = { global.APDC_cooperation_active_law_option = global.APDC_cooperation_law_queue_options^0 }
		set_variable = { global.APDC_cooperation_active_law_proposer = global.APDC_cooperation_law_queue_proposers^0 }
		remove_from_array = { array = global.APDC_cooperation_law_queue_groups index = 0 }
		remove_from_array = { array = global.APDC_cooperation_law_queue_options index = 0 }
		remove_from_array = { array = global.APDC_cooperation_law_queue_proposers index = 0 }

		clear_array = global.APDC_cooperation_vote_supporters
		clear_array = global.APDC_cooperation_vote_abstainers
		clear_array = global.APDC_cooperation_vote_vetoers
		for_each_scope_loop = {
			array = global.APDC_fifth_international_cooperation_members
			if = {
				limit = { tag = var:global.APDC_cooperation_active_law_proposer }
				add_to_array = { array = global.APDC_cooperation_vote_supporters value = THIS }
			}
			else = {
				add_to_array = { array = global.APDC_cooperation_vote_abstainers value = THIS }
			}
		}
		set_variable = { global.APDC_cooperation_vote_state = 1 }
		set_variable = { global.APDC_cooperation_vote_days_remaining = 30 }
	}
}

# APDC_cooperation_law_group_id is supplied by the dynamic law-group entry;
# APDC_cooperation_submitted_option is set by the clicked option button.
APDC_submit_cooperation_law = {
	set_temp_variable = { APDC_cooperation_submitted_group = APDC_cooperation_law_group_id }
	if = {
		limit = {
			APDC_is_fifth_international_cooperation_organization_member = yes
			# Validate every legal option count explicitly for this game build.
			OR = {
				check_variable = { APDC_cooperation_submitted_option = 1 }
				AND = {
					check_variable = { APDC_cooperation_submitted_option = 2 }
					check_variable = { global.APDC_cooperation_law_option_counts^APDC_cooperation_submitted_group > 1 }
				}
				AND = {
					check_variable = { APDC_cooperation_submitted_option = 3 }
					check_variable = { global.APDC_cooperation_law_option_counts^APDC_cooperation_submitted_group > 2 }
				}
				AND = {
					check_variable = { APDC_cooperation_submitted_option = 4 }
					check_variable = { global.APDC_cooperation_law_option_counts^APDC_cooperation_submitted_group > 3 }
				}
			}
			check_variable = { global.APDC_cooperation_law_group_locks^APDC_cooperation_submitted_group = 0 }
			OR = {
				AND = {
					check_variable = { APDC_cooperation_submitted_option = 1 }
					NOT = { check_variable = { global.APDC_cooperation_law_selected_options^APDC_cooperation_submitted_group = 1 } }
				}
				AND = {
					check_variable = { APDC_cooperation_submitted_option = 2 }
					NOT = { check_variable = { global.APDC_cooperation_law_selected_options^APDC_cooperation_submitted_group = 2 } }
				}
				AND = {
					check_variable = { APDC_cooperation_submitted_option = 3 }
					NOT = { check_variable = { global.APDC_cooperation_law_selected_options^APDC_cooperation_submitted_group = 3 } }
				}
				AND = {
					check_variable = { APDC_cooperation_submitted_option = 4 }
					NOT = { check_variable = { global.APDC_cooperation_law_selected_options^APDC_cooperation_submitted_group = 4 } }
				}
			}
			OR = {
				AND = { check_variable = { APDC_cooperation_submitted_option = 1 } check_variable = { global.APDC_cooperation_law_option_1_available^APDC_cooperation_submitted_group = 1 } }
				AND = { check_variable = { APDC_cooperation_submitted_option = 2 } check_variable = { global.APDC_cooperation_law_option_2_available^APDC_cooperation_submitted_group = 1 } }
				AND = { check_variable = { APDC_cooperation_submitted_option = 3 } check_variable = { global.APDC_cooperation_law_option_3_available^APDC_cooperation_submitted_group = 1 } }
				AND = { check_variable = { APDC_cooperation_submitted_option = 4 } check_variable = { global.APDC_cooperation_law_option_4_available^APDC_cooperation_submitted_group = 1 } }
			}
		}
		add_to_array = { array = global.APDC_cooperation_law_queue_groups value = APDC_cooperation_submitted_group }
		add_to_array = { array = global.APDC_cooperation_law_queue_options value = APDC_cooperation_submitted_option }
		add_to_array = { array = global.APDC_cooperation_law_queue_proposers value = THIS }
		set_variable = { global.APDC_cooperation_law_group_locks^APDC_cooperation_submitted_group = 1 }
		APDC_start_next_cooperation_law_vote = yes
	}
}

APDC_resolve_cooperation_law_vote = {
	# Copy the active identifiers to local indices before mutating the global
	# arrays.  Clausewitz reliably accepts temporary variables as array indices.
	set_temp_variable = { APDC_cooperation_resolved_group = global.APDC_cooperation_active_law_group }
	set_temp_variable = { APDC_cooperation_resolved_option = global.APDC_cooperation_active_law_option }
	if = {
		limit = { check_variable = { global.APDC_cooperation_vote_supporters^num > global.APDC_cooperation_vote_vetoers^num } }
		set_variable = { global.APDC_cooperation_law_selected_options^APDC_cooperation_resolved_group = APDC_cooperation_resolved_option }
		add_to_variable = { global.APDC_cooperation_enacted_law_revision = 1 }
		for_each_scope_loop = {
			array = global.APDC_fifth_international_cooperation_members
			set_variable = { APDC_cooperation_last_enacted_law_group = global.APDC_cooperation_active_law_group }
			set_variable = { APDC_cooperation_last_enacted_law_option = global.APDC_cooperation_active_law_option }
			set_variable = { APDC_cooperation_law_effect_revision = global.APDC_cooperation_enacted_law_revision }
			set_country_flag = APDC_cooperation_law_effects_active
		}
	}

	# Passing or failing ends this group's lock; another option may be
	# proposed after the mandatory recess.
	set_variable = { global.APDC_cooperation_law_group_locks^APDC_cooperation_resolved_group = 0 }
	clear_array = global.APDC_cooperation_vote_supporters
	clear_array = global.APDC_cooperation_vote_abstainers
	clear_array = global.APDC_cooperation_vote_vetoers
	clear_variable = global.APDC_cooperation_active_law_group
	clear_variable = global.APDC_cooperation_active_law_option
	clear_variable = global.APDC_cooperation_active_law_proposer
	set_variable = { global.APDC_cooperation_vote_state = 2 }
	set_variable = { global.APDC_cooperation_vote_days_remaining = 30 }
}

APDC_cooperation_legislation_daily_tick = {
	if = {
		limit = { has_global_flag = APDC_cooperation_legislation_initialized_v2 }
		if = {
			limit = { check_variable = { global.APDC_cooperation_vote_state = 1 } }
			add_to_variable = { global.APDC_cooperation_vote_days_remaining = -1 }
			if = {
				limit = { check_variable = { global.APDC_cooperation_vote_days_remaining < 1 } }
				APDC_resolve_cooperation_law_vote = yes
			}
		}
		else_if = {
			limit = { check_variable = { global.APDC_cooperation_vote_state = 2 } }
			add_to_variable = { global.APDC_cooperation_vote_days_remaining = -1 }
			if = {
				limit = { check_variable = { global.APDC_cooperation_vote_days_remaining < 1 } }
				set_variable = { global.APDC_cooperation_vote_state = 0 }
				clear_variable = global.APDC_cooperation_vote_days_remaining
				APDC_start_next_cooperation_law_vote = yes
			}
		}
		else = {
			APDC_start_next_cooperation_law_vote = yes
		}
	}
}
