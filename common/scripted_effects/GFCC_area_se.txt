GFCC_area_Pollution_effects_1 = {
	if = {
		limit = {
			check_variable = { GFCC_area_Pollution_var > 0 }
			# has_dynamic_modifier = { modifier = SLE_OGAS_diqu_2 }
		}
		add_to_variable = { GFCC_area_Pollution_var = -10 }

			min = 0
			max = 99999
		}
	}
}

# 黑区周边辐射
GFCC_area_Pollution_effects_2 = {
	if = {
		limit = {
			any_neighbor_state = {
				has_state_flag = GFCC_area_Pollution_flag #如果有污染标记，则以下成立
				check_variable = { GFCC_area_Pollution_var > 9999 }
			}
		}
		if = {
			limit = {
				check_variable = { GFCC_area_Pollution_var > 149 }
				check_variable = { GFCC_area_Pollution_var < 5000 }
			}
			add_to_variable = { GFCC_area_Pollution_var = 5 }
		}
		if = {
			limit = {
				check_variable = { GFCC_area_Pollution_var > -1 }
				check_variable = { GFCC_area_Pollution_var < 150 }
			}
			add_to_variable = { GFCC_area_Pollution_var = 10 }
		}
		clamp_variable = {
			var = GFCC_area_Pollution_var
			min = 0
			max = 9999
		}
	}
}

#重建度受战争影响：争夺中的地区且基础设施受损时重建度下降
GFCC_area_reconstruction_effects_1 = {
	if = {
		limit = {
			has_state_flag = GFCC_area_reconstruction_flag #如果有重建度标记，则以下成立
			NOT = {
				OR = {
					is_fully_controlled_by = controller
					is_fully_controlled_by = owner
				}
			}
			check_variable = { var = damaged_building_level@infrastructure value = 0 compare = greater_than } #基础设施受损检测
		}
		add_to_variable = {
			var = GFCC_area_reconstruction_var
			value = -3
		}
		clamp_variable = {
			var = GFCC_area_reconstruction_var
			min = 0
			max = 100
		}
	}
}
#重建度增长：完全控制或和平时增长，增长率受安全度影响
GFCC_area_reconstruction_effects_3 = {
	if = {
		limit = {
			has_state_flag = GFCC_area_reconstruction_flag #如果有重建度标记，则以下成立
			OR = {
				is_fully_controlled_by = controller
				is_fully_controlled_by = owner
				controller = { has_war = no }
			}
		}
		#增长率取决于安全度
		if = {
			limit = {
				check_variable = { GFCC_area_security_var > 79 }
			}
			add_to_variable = {
				var = GFCC_area_reconstruction_var
				value = 1
			}
		}
		else_if = {
			limit = {
				check_variable = { GFCC_area_security_var > 59 }
			}
			add_to_variable = {
				var = GFCC_area_reconstruction_var
				value = 0.6
			}
		}
		else_if = {
			limit = {
				check_variable = { GFCC_area_security_var > 39 }
			}
			add_to_variable = {
				var = GFCC_area_reconstruction_var
				value = 0.3
			}
		}
		else = {
			add_to_variable = {
				var = GFCC_area_reconstruction_var
				value = 0.1
			}
		}
		clamp_variable = {
			var = GFCC_area_reconstruction_var
			min = 0
			max = 100
		}
	}
}

		limit = {
			check_variable = { GFCC_area_Pollution_var > 10 } #黄区/蓝区
		}
		add_to_variable = {
			var = GFCC_area_security_var
			value = -0.5
		}
	}
	else_if = {
		limit = {
			check_variable = { GFCC_area_Pollution_var = 0 } #白区
		}
		add_to_variable = {
			var = GFCC_area_security_var
			value = 0.1
		}
	}
	#重建度影响（重建度越低，安全度越低）
	if = {
		limit = {
			check_variable = { GFCC_area_reconstruction_var < 50 }
		}
		add_to_variable = {
			var = GFCC_area_security_var
			value = -0.5
		}
	}
	else_if = {
		limit = {
			check_variable = { GFCC_area_reconstruction_var < 70 }
		}
		add_to_variable = {
			var = GFCC_area_security_var
			value = -0.25
		}
	}
	#战争影响（争夺中的地区安全度下降）
	if = {
		limit = {
			NOT = {
				OR = {
					is_fully_controlled_by = controller
					is_fully_controlled_by = owner
				}
			}
			check_variable = { var = damaged_building_level@infrastructure value = 0 compare = greater_than } #基础设施受损检测
		}
		add_to_variable = {
			var = GFCC_area_security_var
			value = -0.1
		}
	}
	#占领影响（非核心且抵抗度高、顺从度低的地区）
	if = {
		limit = {
			# NOT = { is_core_of = controller }
			resistance > 50
			compliance < 30
		}
		add_to_variable = {
			var = GFCC_area_security_var
			value = -0.1
		}
	}
	#自然恢复（红区以下、重建充分且完全控制时恢复）
	if = {
		limit = {
			check_variable = { GFCC_area_Pollution_var < 150 }
			check_variable = { GFCC_area_reconstruction_var > 69 }
			OR = {
				is_fully_controlled_by = controller
				is_fully_controlled_by = owner
			}
		}
		add_to_variable = {
			var = GFCC_area_security_var
			value = 1
		}
	}
	clamp_variable = {
		var = GFCC_area_security_var
		min = 0
		max = 100
	}
}
